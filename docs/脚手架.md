命令行

```shell
APP_NAME=nest-scaffold

cd ./${APP_NAME}
touch ./.env
rm ./src/app.controller.spec.ts ./src/app.controller.ts ./src/app.service.ts ./src/app.module.ts
nest g mo app
```

# 添加 vitest-base.config.mts

```ts
import { resolve } from 'path';
import swc from 'unplugin-swc';
import { defineConfig } from 'vitest/config';

export default defineConfig({
  test: {
    root: './',
    env: {
      NODE_ENV: 'test',
    },
  },
  plugins: [
    // 使用 SWC 编译测试，保留 Nest 装饰器元数据支持。
    swc.vite({
      // 覆盖 .swcrc 中的模块类型。
      module: { type: 'es6' },
    }),
  ],
  resolve: {
    alias: {
      '@': resolve(import.meta.dirname, './src'),
    },
  },
});
```

共享配置只负责测试根目录、SWC 编译、路径别名和测试环境变量。三个分类配置都只继承共享配置，不依赖聚合入口，避免循环引用。

# 添加 vitest-unit.config.mts

```ts
import { defineConfig } from 'vitest/config';
import baseConfig from './vitest-base.config.mjs';

export default defineConfig({
  ...baseConfig,
  test: {
    ...baseConfig.test,
    name: 'unit',
    include: ['src/**/*.unit-spec.ts'],
    coverage: {
      provider: 'v8',
      reporter: ['text', 'html', 'lcov'],
      thresholds: {
        statements: 40,
        branches: 37,
        functions: 40,
        lines: 40,
      },
      include: ['src/**/*.ts'],
      exclude: [
        'src/**/*.unit-spec.ts',
        'src/**/*.integration-spec.ts',
        'src/**/*.e2e-spec.ts',
        'src/**/__tests__/**',
        'src/**/*.module.ts',
        'src/main.ts',
        'src/**/*.d.ts',
        'src/database/**',
        'src/i18n/**',
      ],
    },
  },
});
```

单元测试放在被测代码附近的 `__tests__/` 中，命名为 `<name>.unit-spec.ts`。`pnpm test:unit`、`pnpm test:unit:watch` 和 `pnpm test:unit:cov` 都只运行单测，覆盖率和阈值仅放在单测配置里。测试 API 从 `vitest` 显式导入。

# 创建 vitest-integration.config.mts

```ts
import { defineConfig } from 'vitest/config';
import baseConfig from './vitest-base.config.mjs';

export default defineConfig({
  ...baseConfig,
  test: {
    ...baseConfig.test,
    name: 'integration',
    include: ['src/**/*.integration-spec.ts'],
    hookTimeout: 120_000,
    testTimeout: 60_000,
  },
});
```

集成测试可以使用 Testcontainers 启动真实 MySQL/Redis，但只要测试仍自行装配模块或替换应用内部依赖，就不属于 E2E。

# 创建 vitest-e2e.config.mts

```ts
import { defineConfig } from 'vitest/config';
import baseConfig from './vitest-base.config.mjs';

export default defineConfig({
  ...baseConfig,
  test: {
    ...baseConfig.test,
    name: 'e2e',
    passWithNoTests: true,
    include: ['test/e2e/**/*.e2e-spec.ts'],
    hookTimeout: 120_000,
    testTimeout: 60_000,
  },
});
```

E2E（端到端测试）必须启动完整应用装配，不替换 Provider/Pipe/Filter/配置，并且通过 HTTP 等公开边界验证。当前预留 E2E 入口，允许空测试集。

# 添加 vitest.config.mts

```ts
import { defineConfig } from 'vitest/config';

export default defineConfig({
  test: {
    projects: [
      { extends: './vitest-unit.config.mts' },
      { extends: './vitest-integration.config.mts' },
      { extends: './vitest-e2e.config.mts' },
    ],
  },
});
```

`pnpm test` 通过此聚合入口运行全部三类测试；集成测试需要启动 Docker，并自动创建和清理临时依赖容器。

CI（持续集成检查）的 `ci` 任务运行单测覆盖率和集成测试；`docker` 任务验证生产镜像构建。

调试单测时显式选择单测配置：`pnpm exec vitest run --config ./vitest-unit.config.mts --inspect-brk --no-file-parallelism --test-timeout=0 <文件路径>`。

### Docker 环境

创建 \docker-compose.yml 文件

```yml
#mysql，redis，phpmyadmin 服务 用于 nestjs 项目
services:
  mysql:
    container_name: ${APP_NAME}-mysql
    image: mysql:9
    command: --character-set-server=utf8mb4 --collation-server=utf8mb4_general_ci
    #    image: mysql:5.7
    #    platform: linux/amd64
    #    command: --default-authentication-plugin=mysql_native_password --character-set-server=utf8mb4 --collation-server=utf8mb4_general_ci
    restart: always
    environment:
      MYSQL_ROOT_PASSWORD: 'root_password'
      MYSQL_DATABASE: ${MYSQL_DATABASE:-dev_db}
    ports:
      - ${MYSQL_PORT:-3306}:3306
    networks:
      - nestjs

  phpmyadmin:
    container_name: ${APP_NAME}-phpmyadmin
    image: phpmyadmin
    restart: always
    ports:
      - '8080:80'
    environment:
      PMA_HOST: mysql # 这里的 db-dev 是 db-dev 服务的名称
      PMA_PORT: ${MYSQL_PORT:-3306}
      PMA_USER: 'root'
      PMA_PASSWORD: 'root_password'
      UPLOAD_LIMIT: 100000000
    depends_on:
      - mysql
    networks:
      - nestjs

  redis:
    container_name: ${APP_NAME}-redis
    image: redis
    command: redis-server --requirepass ${REDIS_PASSWORD:-redis_password}
    restart: always
    ports:
      - ${REDIS_PORT:-6379}:6379
    networks:
      - nestjs

  phpRedisAdmin:
    container_name: ${APP_NAME}-phpRedisAdmin
    image: erikdubbelboer/phpredisadmin
    restart: always
    ports:
      - '8081:80'
    environment:
      REDIS_1_HOST: redis
      REDIS_1_PORT: ${REDIS_PORT:-6379}
      REDIS_1_AUTH: ${REDIS_PASSWORD:-redis_password}
    depends_on:
      - redis
    networks:
      - nestjs

networks:
  nestjs:
    driver: bridge
```

### 启动开发环境

命令行

```shell
# 创建并启动容器 会读取.env 变量创建
docker-compose -p ${APP_NAME} up -d
# 停止并删除容器
docker-compose down
# 启动容器
docker-compose start
# 停止容器
docker-compose stop
```

修改 drizzle.config.ts

```typescript
import 'dotenv/config';
import { defineConfig } from 'drizzle-kit';

export default defineConfig({
  out: './drizzle',
  schema: './src/database/schema.ts',
  dialect: 'mysql',
  dbCredentials: {
    host: process.env.MYSQL_HOST!,
    port: parseInt(process.env.MYSQL_PORT!),
    database: process.env.MYSQL_DATABASE!.replace(
      '${APP_NAME}', // 因为数据库名可能是根据 APP_NAME 动态生成的
      process.env.APP_NAME!,
    ),
    user: process.env.MYSQL_USER,
    password: process.env.MYSQL_PASSWORD,
  },
});
```
